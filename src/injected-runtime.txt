    const leftRotate32 = (value, bits) => (
      (value << bits) | (value >>> (32 - bits))

    const md5Hex = (value) => {
      const bytes = [...new TextEncoder().encode(String(value))];
      const bitLength = bytes.length * 8;
      bytes.push(0x80);
      while (bytes.length % 64 !== 56) {
        bytes.push(0);
      }
      for (let index = 0; index < 8; index += 1) {
        bytes.push(Math.floor(bitLength / (2 ** (8 * index))) & 0xff);
      }

      let a0 = 0x67452301;
      let b0 = 0xefcdab89;
      let c0 = 0x98badcfe;
      let d0 = 0x10325476;
      const shiftAmounts = [
        7, 12, 17, 22, 7, 12, 17, 22, 7, 12, 17, 22, 7, 12, 17, 22,
        5, 9, 14, 20, 5, 9, 14, 20, 5, 9, 14, 20, 5, 9, 14, 20,
        4, 11, 16, 23, 4, 11, 16, 23, 4, 11, 16, 23, 4, 11, 16, 23,
        6, 10, 15, 21, 6, 10, 15, 21, 6, 10, 15, 21, 6, 10, 15, 21
      ];
      const constants = Array.from({ length: 64 }, (_item, index) => (
        Math.floor(Math.abs(Math.sin(index + 1)) * 2 ** 32) | 0
      ));

      for (let offset = 0; offset < bytes.length; offset += 64) {
        const words = [];
        for (let index = 0; index < 16; index += 1) {
          const cursor = offset + (index * 4);
          words[index] = (
            bytes[cursor]
            | (bytes[cursor + 1] << 8)
            | (bytes[cursor + 2] << 16)
            | (bytes[cursor + 3] << 24)
          );
        }

        let a = a0;
        let b = b0;
        let c = c0;
        let d = d0;
        for (let index = 0; index < 64; index += 1) {
          let f;
          let wordIndex;
          if (index < 16) {
            f = (b & c) | ((~b) & d);
            wordIndex = index;
          } else if (index < 32) {
            f = (d & b) | ((~d) & c);
            wordIndex = (5 * index + 1) % 16;
          } else if (index < 48) {
            f = b ^ c ^ d;
            wordIndex = (3 * index + 5) % 16;
          } else {
            f = c ^ (b | (~d));
            wordIndex = (7 * index) % 16;
          }
          const next = d;
          d = c;
          c = b;
          b = (b + leftRotate32((a + f + constants[index] + words[wordIndex]) | 0, shiftAmounts[index])) | 0;
          a = next;
        }

        a0 = (a0 + a) | 0;
        b0 = (b0 + b) | 0;
        c0 = (c0 + c) | 0;
        d0 = (d0 + d) | 0;
      }

      const wordToHex = (word) => [0, 8, 16, 24]
        .map((shift) => ((word >>> shift) & 0xff).toString(16).padStart(2, "0"))
        .join("");
      return [a0, b0, c0, d0].map(wordToHex).join("");
    };

          clearTimeout(timer);
          timer = null;

          return;

        cleanup();
        if (data.ok && typeof data.translated === "string" && data.translated.trim()) {
          resolve(data.translated.trim());
          return;
        }
        reject(new Error(data.error || "Baidu Translate bridge returned no result."));

        cleanup();
        reject(new Error("Baidu Translate bridge timed out."));

        source: requestMessage,
        id,
        payload: { text: String(text || "").trim() }

    const translateWithBaidu = async (text) => {

            appId: config.appId,
            secretKey: config.secretKey
          });

            recordTranslateAttempt("baidu-direct-bridge-ok");
            return translated.trim();

          throw new Error("Baidu Translate bridge returned no result.");
        } catch (error) {
          const message = errorMessage(error);
          errors.push(`direct bridge: ${message}`);
          recordTranslateAttempt("baidu-direct-bridge-failed", { error: message });
        }
      } else {
        recordTranslateAttempt("baidu-direct-bridge-missing");
      }
      try {
        recordTranslateAttempt("baidu-window-bridge");

          recordTranslateAttempt("baidu-window-bridge-ok");
          return translated.trim();

        const message = errorMessage(error);
        errors.push(`window bridge: ${message}`);
        recordTranslateAttempt("baidu-window-bridge-failed", { error: message });
      }
      try {
        recordTranslateAttempt("baidu-renderer-fetch");
        const salt = `${Date.now()}${Math.floor(Math.random() * 100000)}`;
        const params = new URLSearchParams({
          q: query,
          from: "auto",
          to: "zh",
          appid: config.appId,
          salt,
          sign: md5Hex(`${config.appId}${query}${salt}${config.secretKey}`)

          method: "GET",
          credentials: "omit"
        });
        if (!response.ok) {
          throw new Error(`Baidu Translate HTTP ${response.status}`);
        }
        const payload = await response.json();
        if (payload?.error_code) {
          throw new Error(payload.error_msg || `Baidu Translate error ${payload.error_code}`);
        }
        const lines = Array.isArray(payload?.trans_result)
          ? payload.trans_result.map((item) => item?.dst).filter(Boolean)
          : [];
        if (lines.length === 0) {
          throw new Error("Baidu Translate returned no result.");
        }
        recordTranslateAttempt("baidu-renderer-fetch-ok");
        return lines.join("\n");
      } catch (error) {
        const message = errorMessage(error);
        errors.push(`renderer fetch: ${message}`);
        recordTranslateAttempt("baidu-renderer-fetch-failed", { error: message });
      }
      throw new Error(errors.join("; ") || "Baidu Translate failed.");
    };

    const escapeRegExp = (value) => value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

    const compilePlaceholderPattern = (source, target) => {
      if (!source.includes("{") || source.includes("plural,")) {
        return null;
      }

      const literalText = source.replace(/\{[a-zA-Z0-9_]+\}/g, "");
      if (literalText.replace(/\s+/g, "").length < 4) {
        return null;
      }

      const names = [];
      const pattern = escapeRegExp(source).replace(/\\\{([a-zA-Z0-9_]+)\\\}/g, (_, name) => {
        names.push(name);
        return "(.+?)";
      });

      if (names.length === 0) {
        return null;
      }

      return {
        regex: new RegExp(`^${pattern}$`),
        render(match) {
          let rendered = target;
          names.forEach((name, index) => {
            rendered = rendered.replaceAll(`{${name}}`, match[index + 1]);
          });
          return rendered;
        }
      };

      if (connectionNeedsMatch) {
        return `${leading}连接还需要 ${connectionNeedsMatch[1]} 个字段${trailing}`;

        for (const [source, target] of state.segmentEntries) {
          if (rendered.includes(source)) {
            rendered = rendered.replaceAll(source, target);
          }
        }
        if (rendered !== body) {

        }

      "white-space:nowrap"
    ].join(";");

    const translateSkillDescriptionElement = async (target, button) => {
      const original = normalizedSkillTranslationSource(target.textContent || "");
      if (!original) {
        return false;
      }
      recordSkillTranslateDebug({ phase: "start", text: original });
      const cached = cachedSkillTranslation(original);
      if (cached) {
        target.setAttribute(skillTranslateTargetMarker, "true");
        target.setAttribute("data-claude-cn-skip", "true");
        target.removeAttribute("data-claude-ultra-skill-auto");
        target.removeAttribute("data-claude-ultra-skill-error");
        target.textContent = cached;
        recordSkillTranslateDebug({ phase: "cached", text: original });
        if (button) {
          button.textContent = ultraText("translatedSkillDescription");
        }
        return true;
      }
      if (button) {
        button.disabled = true;
        button.style.cursor = "default";
        button.textContent = ultraText("translatingSkillDescription");
      }
      try {
        const translated = await translateWithBaidu(original);
        rememberSkillTranslation(original, translated);
        target.setAttribute(skillTranslateTargetMarker, "true");
        target.setAttribute("data-claude-cn-skip", "true");
        target.removeAttribute("data-claude-ultra-skill-auto");
        target.removeAttribute("data-claude-ultra-skill-error");
        target.textContent = translated;
        recordSkillTranslateDebug({ phase: "translated", text: original });
        if (button) {
          button.textContent = ultraText("translatedSkillDescription");
        }
        return true;
      } catch (error) {
        const errorText = String(error?.message || error || ultraText("translateFailed"));
        target.setAttribute("data-claude-ultra-skill-error", errorText.slice(0, 300));
        recordSkillTranslateDebug({ phase: "failed", text: original, error: errorText });
        if (button) {
          button.disabled = false;
          button.style.cursor = "pointer";
          button.textContent = ultraText("translateFailed");
          button.title = errorText || ultraText("translateFailed");
          setTimeout(() => {
            if (button.isConnected) {
              button.textContent = ultraText("translateSkillDescription");
            }
          }, 2200);
        }
        return false;
      }
    };

    const autoTranslateSkillDescriptionTarget = async (target) => {
      if (state.skillAutoTranslatePending.has(target) || target.getAttribute("data-claude-ultra-skill-auto") === "failed") {
        return;
      }
      state.skillAutoTranslatePending.add(target);
      target.setAttribute("data-claude-ultra-skill-auto", "pending");
      const translated = await translateSkillDescriptionElement(target, null);
      state.skillAutoTranslatePending.delete(target);
      if (!translated && target.isConnected && target.getAttribute("data-claude-ultra-skill-auto") === "pending") {
        target.setAttribute("data-claude-ultra-skill-auto", "failed");
      }
    };

    const ensureSkillDescriptionTranslateButton = (target) => {
      const section = target.parentElement;
      const heading = section?.querySelector("h2,h3,h4,[role='heading']");
      const headingRow = heading?.closest("div") || section;
      if (!headingRow) {
        return;
      }

      let button = [...document.querySelectorAll(`[${skillTranslateMarker}]`)]
        .find((candidate) => candidate.__claudeUltraSkillTranslateTarget === target);
      if (!button) {
        button = document.createElement("button");
        button.type = "button";
        button.setAttribute(skillTranslateMarker, "true");
        button.setAttribute("data-claude-cn-skip", "true");
        button.style.cssText = skillTranslateButtonStyle();
        button.addEventListener("click", (event) => {
          event.preventDefault();
          event.stopPropagation();
          translateSkillDescriptionElement(target, button);
        });
        button.addEventListener("keydown", (event) => {
          event.stopPropagation();
        });
      }

      button.__claudeUltraSkillTranslateTarget = target;
      const label = ultraText("translateSkillDescription");
      if (!button.disabled && button.textContent !== label) {
        button.textContent = label;
      }
      if (button.title !== label) {
        button.title = label;
      }
      if (!button.isConnected || button.parentElement !== headingRow) {
        headingRow.appendChild(button);
      }
    };

    const renderSkillDescriptionTranslateButtons = () => {
      if (!document.body) {
        return;
      }
      const configured = hasBaiduTranslateConfig();
      const autoTranslate = isBaiduAutoTranslateEnabled();
      const detailTargets = configured ? new Set(findSkillDescriptionTargets()) : new Set();
      const slashHoverTargets = autoTranslate ? new Set(findSlashSkillHoverDescriptionTargets()) : new Set();
      for (const button of document.querySelectorAll(`[${skillTranslateMarker}]`)) {
        const target = button.__claudeUltraSkillTranslateTarget;
        if (!target || !target.isConnected || !detailTargets.has(target) || autoTranslate) {
          button.remove();
        }
      }
      for (const target of detailTargets) {
        if (autoTranslate) {
          autoTranslateSkillDescriptionTarget(target);
        } else {
          ensureSkillDescriptionTranslateButton(target);
        }
      }
      for (const target of slashHoverTargets) {
        autoTranslateSkillDescriptionTarget(target);
      }
    };

    const processRootInternal = (root) => {
      if (root.nodeType === Node.TEXT_NODE) {
        processTextNode(root);
        return;
      }

      if (root.nodeType !== Node.ELEMENT_NODE && root.nodeType !== Node.DOCUMENT_NODE && root.nodeType !== Node.DOCUMENT_FRAGMENT_NODE) {
        return;
      }

      if (root.nodeType === Node.ELEMENT_NODE) {
        processElement(root);
        if (root.shadowRoot) {
          processRootInternal(root.shadowRoot);
        }
      }

      const walker = document.createTreeWalker(
        root,
        NodeFilter.SHOW_ELEMENT | NodeFilter.SHOW_TEXT
      );

      for (let node = walker.nextNode(); node; node = walker.nextNode()) {
        if (node.nodeType === Node.ELEMENT_NODE) {
          processElement(node);
          if (node.shadowRoot) {
            processRootInternal(node.shadowRoot);
          }
        } else if (node.nodeType === Node.TEXT_NODE) {
          processTextNode(node);
        }
      }

      if (state.enabled) {
        document.title = translate(document.title);
      }
    };

    const processRoot = (root) => {
      withMutationPause(() => {
        syncEnabled();
        processRootInternal(root);
        renderLanguageCard();
        renderUltraMenu();
        renderUltraPanel();
        renderSkillDescriptionTranslateButtons();
      });
    };

    const refresh = () => {
      if (document.documentElement) {
        processRoot(document.documentElement);
      } else {
        renderLanguageCard();
        renderUltraMenu();
        renderUltraPanel();
        renderSkillDescriptionTranslateButtons();
      }
    };

    const scheduleRefresh = (delay = 60) => {
      if (state.refreshTimer !== null) {
        clearTimeout(state.refreshTimer);
      }
      state.refreshTimer = setTimeout(() => {
        state.refreshTimer = null;
        refresh();
      }, delay);
    };

    const setLocale = (locale) => {
      state.currentLocale = locale;
      refresh();
      scheduleRefresh(400);
    };

    const requestLocale = (locale, options = {}) => {
      if (options.persist !== false) {
        writeStoredLocale(locale);
      }
      setLocale(locale);
      const desktopIntl = globalThis["claude.hybrid"]?.DesktopIntl;
      if (desktopIntl?.requestLocaleChange) {
        Promise.resolve(desktopIntl.requestLocaleChange(locale)).catch(() => setLocale(readCurrentLocale()));
      }
    };

    const hookNavigation = () => {
      for (const method of ["pushState", "replaceState"]) {
        const original = history[method];
        if (original?.__claudeUltraPatched) {
          continue;
        }
        const patched = function patchedHistoryMethod(...args) {
          const result = original.apply(this, args);
          setTimeout(refresh, 0);
          scheduleRefresh(600);
          return result;
        };
        patched.__claudeUltraPatched = true;
        history[method] = patched;
      }
      window.addEventListener("popstate", refresh);
    };

    const hookLocaleChanges = () => {
      const desktopIntl = globalThis["claude.hybrid"]?.DesktopIntl;
      desktopIntl?.onLocaleChanged?.((locale) => {
        writeStoredLocale(locale);
        setLocale(locale);
      });
    };

    const hookMessagesFetch = () => {
      const originalFetch = window.fetch;
      if (!originalFetch || originalFetch.__claudeUltraPatched) {
        return;
      }
      const patchedFetch = function patchedClaudeUltraFetch(input, init) {
        try {
          const url = input instanceof Request ? input.url : input;
          const nextInit = prepareMessagesRequestInit(url, init);
          if (nextInit !== init) {
            return originalFetch.call(this, input, nextInit);
          }
        } catch {}
        return originalFetch.call(this, input, init);
      };
      patchedFetch.__claudeUltraPatched = true;
      window.fetch = patchedFetch;
    };

    const start = () => {
      update(config);
      exposeUltraApi();
      state.currentLocale = readCurrentLocale();
      hookMessagesFetch();
      hookNavigation();
      hookLocaleChanges();
      const desiredLocale = preferredLocale();
      if (normalizeLocale(desiredLocale) !== normalizeLocale(state.currentLocale)) {
        requestLocale(desiredLocale, { persist: false });
      } else {
        refresh();
      }
      for (const delay of [250, 1000, 2500, 5000]) {
        setTimeout(refresh, delay);
      }

      state.observer = new MutationObserver((records) => {
        const externalRecords = records.filter((record) => !isOverlayMutationRecord(record));
        if (externalRecords.length === 0) {
          return;
        }
        if (state.mutating) {
          scheduleRefresh(80);
          return;
        }

        withMutationPause(() => {
          syncEnabled();
          for (const record of externalRecords) {
            if (record.type === "characterData") {
              processRootInternal(record.target);
            } else if (record.type === "attributes") {
              processRootInternal(record.target);
            } else {
              for (const node of record.addedNodes) {
                processRootInternal(node);
              }
            }
          }
          renderLanguageCard();
          renderUltraMenu();
          renderUltraPanel();
          renderSkillDescriptionTranslateButtons();
        });
      });

      state.observer.observe(document.documentElement || document, {
        subtree: true,
        childList: true,
        characterData: true,
        attributes: true,
        attributeFilter: state.config.translateAttributes || undefined
      });
    };

    return { update, refresh, start, setLocale, requestLocale };
  };

  if (window[globalKey]) {
    window[globalKey].update(config);
    window[globalKey].refresh();
    return;
  }

  const overlay = createOverlay();
  window[globalKey] = overlay;

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", () => overlay.start(), { once: true });
  } else {
    overlay.start();
  }
}
